CE1-RED
================

ip prefix-list Cust-RED-Networks seq 1 permit 50.0.1.0/24
ip prefix-list Cust-RED-Networks seq 2 permit 50.0.2.0/24
!
route-map ALLOW-RED-CUSTOMERS permit 10
match ip address prefix-list Cust-RED-Networks
exit
!
interface ens4
ip address 11.0.0.2/24
exit
!
interface ens5
ip address 50.0.1.1/24
exit
!
router bgp 201
neighbor 11.0.0.1 remote-as 100
!
address-family ipv4 unicast
network 50.0.1.0/24
neighbor 11.0.0.1 route-map ALLOW-RED-CUSTOMERS in
neighbor 11.0.0.1 route-map ALLOW-RED-CUSTOMERS out
exit-address-family
exit
!
end
